class Solution {
    public int[] deckRevealedIncreasing(int[] deck) {
        Arrays.sort(deck);
        Queue<Integer> q = new LinkedList<>();
        for(int i=0;i<deck.length;i++)q.offer(i);

        int ans[]=new int[deck.length];
        for(int c : deck){
            ans[q.poll()]=c;
            if(!q.isEmpty()){
                q.add(q.poll());
            }
        }  
        return ans; 
    }
}
