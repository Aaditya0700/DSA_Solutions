class Solution {
    public int longestConsecutive(int[] nums) {
        int max = 0;
        Set<Integer> set = new HashSet<>();
        int len = 0;
        for (int i : nums)
            set.add(i);
        for (int i : set) {
            if (!set.contains(i-1)) {
                int curr = i;
                len = 1;
                while (set.contains(curr + 1)) {
                    curr++;
                    len++;
                }
                max = Math.max(max, len);
            }
        }
        return max;
    }
}
