class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        int n=nums1.length;
        int m=nums2.length;
        int ans[]=new int[m+n];
        int i=0,j=0,idx=0;
        while(i<n && j<m){
            if(nums1[i]<=nums2[j]){
                ans[idx++]=nums1[i++];
            }else{
                ans[idx++]=nums2[j++];
            }
        }
        while(i<n){
            ans[idx++]=nums1[i++];
        }
        while(j<m){
            ans[idx++]=nums2[j++];
        }
        int mid=(m+n)/2;
        if((m+n)%2==0){
            return (double)(((double)ans[mid]+ans[mid-1])/2);
        }else{
            return (double)(ans[(m+n)/2]);
        }
    }
}
