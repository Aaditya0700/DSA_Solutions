class Solution {
    public int longestSubsequence(int[] nums) {
        int sum=nums[0];
        int xor=nums[0];
        for(int i=1;i<nums.length;i++){
            xor^=nums[i];
            sum+=nums[i];
        }
        if(xor!=0)return nums.length;
        if(sum==0)return 0;
        return nums.length-1;
    }
}
