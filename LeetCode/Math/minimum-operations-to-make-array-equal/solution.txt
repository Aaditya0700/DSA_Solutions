class Solution {
    public int minOperations(int n) {
        int ans []=new int [n];
        for(int i=0;i<n;i++){
            ans[i]=2*i+1;
        }
        int mid=ans[0]+(ans[ans.length-1]-ans[0])/2;
        int sum=0;
        for(int i=0;i<ans.length/2;i++){
            sum+=(Math.abs(mid-ans[i]));
        }
        return sum;
    }
}
