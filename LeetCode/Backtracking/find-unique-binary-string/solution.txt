class Solution {
    void solve(List<String> ans,StringBuilder s,int n){
        if(s.length()==n){
            ans.add(s.toString());
            return;
        }
        s.append('0');
        solve(ans,s,n);
        s.deleteCharAt(s.length()-1);

        s.append('1');
        solve(ans,s,n);
        s.deleteCharAt(s.length()-1);
    }
    public String findDifferentBinaryString(String[] nums) {
        int n =nums[0].length();
        List<String> ans = new ArrayList<>();
        StringBuilder s=new StringBuilder();
        solve(ans,s,n);
        Set<String> set=new HashSet<>();
        for(String c : nums)set.add(c);
        for(String c : ans){
            if(!set.contains(c))return  c;
        }
        return "";
    }
}
