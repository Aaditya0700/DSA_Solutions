class Solution {
    static void solve(List<String> ans,StringBuilder s,int n){
        if(s.length()==n){
            ans.add(s.toString());
            return;
        }

        s.append('1');
        solve(ans,s,n);
        s.deleteCharAt(s.length()-1);

        if(s.length()==0 || s.charAt(s.length()-1)=='1'){
            s.append('0');
            solve(ans,s,n);
            s.deleteCharAt(s.length()-1);
        }
    }
    public List<String> validStrings(int n) {
        List<String> ans = new ArrayList<>();
        StringBuilder s = new StringBuilder();

        solve(ans,s,n);
        return ans;
    }
}
