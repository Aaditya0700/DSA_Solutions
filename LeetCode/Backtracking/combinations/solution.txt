class Solution {
    void recursion(List<List<Integer>> ans ,List<Integer> list,int idx,int k,int n){
        if(list.size()==k){
            ans.add(new ArrayList(list));
            return;
        }
        if(idx>n)return;

        for(int i=idx;i<=n;i++){
            list.add(i);
            recursion(ans,list,i+1,k,n);
            list.remove(list.size()-1);
        }
    }
    public List<List<Integer>> combine(int n, int k) {
        List<List<Integer>> ans = new ArrayList<>();
        List<Integer> list = new ArrayList<>();
        recursion(ans,list,1,k,n);
        return ans;
    }
}
