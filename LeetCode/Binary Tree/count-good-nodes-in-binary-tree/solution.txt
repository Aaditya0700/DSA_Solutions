class Solution {
    int count=0;
    void solve(TreeNode root,int max){
        if(root==null)return;
        solve(root.left,Math.max(max,root.val));
        solve(root.right,Math.max(max,root.val));
        if(root.val>=max)count++;
    }
    public int goodNodes(TreeNode root) {
        solve(root,root.val);
        return count;
    }
}
