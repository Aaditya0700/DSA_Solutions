class Solution {
    public int[] productExceptSelf(int[] nums) {
        int n = nums.length;
        int pre[] = new int [n];
        int post[]= new int [n];
        pre[0]=1;
        post[n-1]=1;
        int pr=1,po=1;
        for(int i=1;i<n;i++){
             pr *= nums[i-1];
             pre[i]=pr;
        }
        for(int i=n-2;i>=0;i--){
             po *= nums[i+1];
             post[i]=po;
        }
        int ans [] = new int[n];
        for(int i=0;i<n;i++){
            ans[i]=pre[i]*post[i];
        }
        return ans;
    }
}
