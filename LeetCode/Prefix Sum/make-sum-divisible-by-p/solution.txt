class Solution {
    public int minSubarray(int[] nums, int p) {
        long sum=0;
        for(int i:nums) sum+=i;
        if(sum%p==0)return 0;
        HashMap<Integer,Integer> map=new HashMap<>();
        map.put(0,-1);
        int r=(int)(sum%p);
        
        int min=nums.length;
         sum=0;
        for(int i=0;i<nums.length;i++){
            sum+=nums[i];
            int curr=(int)(sum%p);
            int req=(curr-r+p)%p;
            if(map.containsKey(req)){
                min=Math.min(min,i-map.get(req));
            }
            map.put(curr,i);
        }
        if(min==nums.length)return -1;
        return min;
    }
}
