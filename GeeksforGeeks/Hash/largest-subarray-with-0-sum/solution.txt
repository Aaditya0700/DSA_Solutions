class Solution {
    int maxLength(int arr[]) {
        Map<Integer,Integer> map=new TreeMap<>();
        map.put(0,-1);
        int sum=0;
        int max=0;
        for(int i=0;i<arr.length;i++){
            sum+=arr[i];
            if(map.containsKey(sum)){
                max=Math.max(max,i-map.get(sum));
            }else{
                map.put(sum,i);
            }
        }
        return max;
    }
}
